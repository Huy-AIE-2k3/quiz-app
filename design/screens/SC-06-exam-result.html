// Vector3 cấu trúc biểu diễn tọa độ không gian
struct Vec3 { double x, y, z; };
Vec3 projectPointOnLine(Vec3 A, Vec3 M0, Vec3 u) {
double t = ((A.x - M0.x)*u.x + (A.y - M0.y)*u.y + (A.z - M0.z)*u.z) / (u.x*u.x + u.y*u.y + u.z*u.z);
return { M0.x + t * u.x, M0.y + t * u.y, M0.z + t * u.z };
}
